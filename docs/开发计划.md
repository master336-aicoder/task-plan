# 任务计划工具箱 - 开发计划

> **For agentic workers:** 本计划采用细粒度任务分解，每个任务均可独立执行并测试。

## 项目概述

### 目标
构建一个功能完善的任务计划管理系统，支持任务创建、调度、执行、监控与告警等功能。

### 架构
- 前后端分离架构
- RESTful API 设计
- 前端：Vue3 + TypeScript + Vite + Element Plus
- 后端：Spring Boot + Quartz + Redis + Flowable

### 技术栈
**前端：**
- Vue 3.2.45
- Vite 4.0.1
- Element Plus 2.2.26
- TypeScript 4.9.4
- Pinia 2.0.28
- Vue Router 4.1.6
- UnoCSS 0.58.9

**后端：**
- Spring Boot
- Quartz (调度引擎)
- Redis (缓存/分布式锁)
- Flowable (工作流)
- Lombok

```java
package com.taskplan.scheduler;

import com.taskplan.entity.Task;
import com.taskplan.entity.TaskInstance;
import com.taskplan.service.TaskInstanceService;
import com.taskplan.service.TaskService;
import lombok.extern.slf4j.Slf4j;
import org.quartz.JobExecutionContext;
import org.quartz.JobExecutionException;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.scheduling.quartz.QuartzJobBean;
import org.springframework.stereotype.Component;

@Slf4j
@Component
public class TaskJob extends QuartzJobBean {

    @Autowired
    private TaskService taskService;

    @Autowired
    private TaskInstanceService taskInstanceService;

    @Override
    protected void executeInternal(JobExecutionContext context) throws JobExecutionException {
        Long taskId = context.getJobDetail().getJobDataMap().getLong("taskId");
        Long scheduleId = context.getJobDetail().getJobDataMap().getLong("scheduleId");

        log.info("开始执行任务: taskId={}, scheduleId={}", taskId, scheduleId);

        TaskInstance instance = null;
        try {
            instance = taskInstanceService.createInstance(taskId, scheduleId, "SCHEDULED");
            taskInstanceService.updateStatus(instance.getId(), "RUNNING");

            Task task = taskService.getTask(taskId);

            String result = executeTask(task);

            taskInstanceService.completeInstance(instance.getId(), "SUCCESS", result, null);
            log.info("任务执行成功: taskId={}, instanceId={}", taskId, instance.getId());
        } catch (Exception e) {
            log.error("任务执行失败: taskId={}", taskId, e);
            if (instance != null) {
                taskInstanceService.completeInstance(instance.getId(), "FAILED", null, e.getMessage());
            }
            throw new JobExecutionException(e);
        }
    }

    private String executeTask(Task task) {
        log.info("执行命令: {}", task.getCommand());
        return "任务执行完成: " + task.getName();
    }
}
```

```java
    private Scheduler scheduler;

    public void scheduleJob(TaskSchedule schedule) throws SchedulerException {
        JobDetail jobDetail = JobBuilder.newJob(TaskJob.class)
                .withIdentity("job_" + schedule.getId(), "task_group")
                .usingJobData("taskId", schedule.getTaskId())
                .usingJobData("scheduleId", schedule.getId())
                .storeDurably()
                .build();

        Trigger trigger = buildTrigger(schedule);

        scheduler.scheduleJob(jobDetail, trigger);
        log.info("调度任务已创建: scheduleId={}", schedule.getId());
    }

    public void unscheduleJob(Long scheduleId) throws SchedulerException {
        TriggerKey triggerKey = TriggerKey.triggerKey("trigger_" + scheduleId, "task_group");
        scheduler.unscheduleJob(triggerKey);
        log.info("调度任务已删除: scheduleId={}", scheduleId);
    }

    public void pauseJob(Long scheduleId) throws SchedulerException {
        JobKey jobKey = JobKey.jobKey("job_" + scheduleId, "task_group");
        scheduler.pauseJob(jobKey);
        log.info("调度任务已暂停: scheduleId={}", scheduleId);
    }

    public void resumeJob(Long scheduleId) throws SchedulerException {
        JobKey jobKey = JobKey.jobKey("job_" + scheduleId, "task_group");
        scheduler.resumeJob(jobKey);
        log.info("调度任务已恢复: scheduleId={}", scheduleId);
    }

    private Trigger buildTrigger(TaskSchedule schedule) {
        TriggerBuilder<Trigger> triggerBuilder = TriggerBuilder.newTrigger()
                .withIdentity("trigger_" + schedule.getId(), "task_group");

        if ("CRON".equals(schedule.getScheduleType())) {
            triggerBuilder.withSchedule(CronScheduleBuilder.cronSchedule(schedule.getCronExpression()));
        } else if ("FIXED_INTERVAL".equals(schedule.getScheduleType())) {
            SimpleScheduleBuilder scheduleBuilder = SimpleScheduleBuilder.simpleSchedule()
                    .withIntervalInMilliseconds(getIntervalMillis(schedule))
                    .repeatForever();
            triggerBuilder.withSchedule(scheduleBuilder);
        }

        return triggerBuilder.build();
    }

    private long getIntervalMillis(TaskSchedule schedule) {
        long interval = schedule.getFixedInterval() != null ? schedule.getFixedInterval() : 60000;
        String unit = schedule.getTimeUnit() != null ? schedule.getTimeUnit() : "SECONDS";

        switch (unit) {
            case "SECONDS": return interval * 1000;
            case "MINUTES": return interval * 60 * 1000;
            case "HOURS": return interval * 60 * 60 * 1000;
            default: return interval;
        }
    }
}
```

```java
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Slf4j
@Service
public class TaskScheduleService {

    @Autowired
    private TaskScheduleRepository scheduleRepository;

    @Autowired
    private SchedulerService schedulerService;

    @Transactional
    public TaskSchedule createSchedule(TaskSchedule schedule) {
        schedule = scheduleRepository.save(schedule);
        try {
            if (schedule.getEnabled() == 1) {
                schedulerService.scheduleJob(schedule);
            }
        } catch (Exception e) {
            log.error("创建调度失败", e);
            throw new BusinessException("创建调度失败");
        }
        return schedule;
    }

    @Transactional
    public TaskSchedule updateSchedule(Long id, TaskSchedule schedule) {
        TaskSchedule existing = getSchedule(id);
        try {
            schedulerService.unscheduleJob(id);
        } catch (Exception e) {
            log.warn("取消旧调度失败", e);
        }
        existing.setScheduleType(schedule.getScheduleType());
        existing.setCronExpression(schedule.getCronExpression());
        existing.setFixedInterval(schedule.getFixedInterval());
        existing.setTimeUnit(schedule.getTimeUnit());
        existing.setEnabled(schedule.getEnabled());
        existing = scheduleRepository.save(existing);
        try {
            if (existing.getEnabled() == 1) {
                schedulerService.scheduleJob(existing);
            }
        } catch (Exception e) {
            log.error("更新调度失败", e);
            throw new BusinessException("更新调度失败");
        }
        return existing;
    }

    @Transactional
    public void deleteSchedule(Long id) {
        try {
            schedulerService.unscheduleJob(id);
        } catch (Exception e) {
            log.warn("取消调度失败", e);
        }
        scheduleRepository.deleteById(id);
    }

    @Transactional
    public void toggleSchedule(Long id, Integer enabled) {
        TaskSchedule schedule = getSchedule(id);
        try {
            if (enabled == 1) {
                schedulerService.resumeJob(id);
            } else {
                schedulerService.pauseJob(id);
            }
        } catch (Exception e) {
            log.error("切换调度状态失败", e);
            throw new BusinessException("切换调度状态失败");
        }
        schedule.setEnabled(enabled);
        scheduleRepository.save(schedule);
    }

    public TaskSchedule getSchedule(Long id) {
        return scheduleRepository.findById(id)
                .orElseThrow(() -> new BusinessException("调度不存在"));
    }

    public List<TaskSchedule> listSchedules(Long taskId) {
        if (taskId != null) {
            return scheduleRepository.findByTaskIdAndEnabled(taskId, 1);
        }
        return scheduleRepository.findAll();
    }
```

```java
        if (instance != null) {
            instance.setStatus(status);
            if ("RUNNING".equals(status)) {
                instance.setStartTime(LocalDateTime.now());
            }
            instanceRepository.save(instance);
        }
    }

    public void completeInstance(Long id, String status, String result, String error) {
        TaskInstance instance = instanceRepository.findById(id).orElse(null);
        if (instance != null) {
            instance.setStatus(status);
            instance.setEndTime(LocalDateTime.now());
            if (instance.getStartTime() != null) {
                instance.setDuration(java.time.Duration.between(instance.getStartTime(), instance.getEndTime()).toMillis());
            }
            instance.setResult(result);
            if (error != null) {
                instance.setLogs(error);
            }
            instanceRepository.save(instance);
        }
    }

    public List<TaskInstance> listInstances(Long taskId) {
        return instanceRepository.findByTaskIdOrderByCreatedTimeDesc(taskId);
    }
```
